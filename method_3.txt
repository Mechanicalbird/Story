! ===================================================================
! ====== Stable Microplane Model with Density Scaling Tables   ======
! ====== Metric Units (MPa) version - Linear Element 215       ======
! ===================================================================

/PREP7

! --- Step 1: Base Parameters ---
E_initial = 30000           ! Initial Young's Modulus, MPa
nu_base = 0.2               ! Poisson's Ratio, unitless
rho_initial = 1.8e-9        ! Initial bulk density (tonne/mm^3)

! --- Step 2: Define Microplane/DPC Material Constants ---
fc = 41             
fb = 48             
ft = 4              
Rt = 1              
D = 20e4            
sigma_cv = -32      
Rc = 2              
gamma_t = 0         
gamma_c = 5e-5      
beta_t = 6000       
beta_c = 4000       

c = 650             
m = 2               
element = 215       

! --- Step 3: Configure Density-Changing Field Variable ---
! Using 1 dependency (Field Variable 1) to natively dynamically adjust E and Density
MP,EX,conc,E_initial
MP,NUXY,conc,nu_base
MP,DENS,conc,rho_initial  

! Assign base Microplane properties
TB,MPLANE,conc,,,DPC
TBDATA,1,fc,fb,ft,Rt,D,sigma_cv
TBDATA,7,Rc,gamma_t,gamma_c,beta_t,beta_c

TB,MPLANE,conc,,,NLOCAL
TBDATA,1,c,m

! --- Step 4: Configure Mesh & Linear Element Settings ---
itype = conc
et,itype,element
keyopt,itype,18,1     ! Set to 1 for Linear CPT215 Element

esel,s,mat,,conc
emodif,all,type,itype
allsel

! --- Hand back control natively to the solver pipeline ---
/SOLU
NLGEOM,ON
NSUBST,50,500,20      ! Forces Workbench to use small structural increments natively
